"""حارس الانهيار الموحّد لكل مدققات النظام — ك-133.

قرار صاحب القرار (2026-10-23): لا تُستدلّ حادثة الانهيار من نصّ المخرج، بل من رمز خروج
مخصّص. كل مدقق يمرّ عبر `run(main)`، فأي خطأ غير متوقّع — ولو بلا نص — يخرج بالرمز 70،

الاستعمال في نهاية كل مدقق:

        from _guard import run      # أو: from validators._guard import run
        run(main)
"""
from __future__ import annotations

import sys
import traceback

CRASH_RC = 70  # رمز الانهيار المعتمد — يمنع التسليم في tools/final_gate.sh


def guard(main) -> int:
    """يشغّل main ويعيد رمز خروج معتمدًا: نتيجة المدقق أو رمز الانهيار 70."""
    try:
        rc = main()
        return 0 if rc is None else int(rc)
    except SystemExit as e:                      # مدقق أنهى نفسه صراحةً
        code = e.code
        return 0 if code is None else (code if isinstance(code, int) else 1)
    except KeyboardInterrupt:
        print(f"توقّف المدقق بطلب المستخدم (رمز {CRASH_RC})", file=sys.stderr)
        return CRASH_RC
    except BaseException as e:                   # أي انهيار آخر — بلا ابتلاع
        print(f"انهيار المدقق (رمز {CRASH_RC}): {type(e).__name__}: {e}", file=sys.stderr)
        traceback.print_exc()
        return CRASH_RC


def run(main) -> None:
    """نقطة الدخول: ينهي العملية برمز معتمد (نتيجة المدقق أو 70 عند الانهيار)."""
    sys.exit(guard(main))

def normalize_word(word: str) -> str:
    return "".join(ch for ch in unicodedata.normalize("NFC", word) if ch not in ARABIC_DIACRITICS)


def strip_markup(text: str) -> str:
    text = re.sub(r"<!--.*?-->", " ", text, flags=re.S)
    text = re.sub(r"`[^`]*`", " ", text)
    text = re.sub(r"!?(?:\[[^\]]*\])\([^)]*\)", " ", text)
    text = re.sub(r"[*_>#|]", " ", text)
    return text


def load_project_dictionary() -> set[str]:
    """يبني قاموسًا من الملفات الموجودة فعليًا، لا من قائمة مخترعة."""
    paths = [ROOT / "appendices" / "terminology.md", ROOT / "appendices" / "style-blacklist.json"]
    paths.extend(sorted((ROOT / "appendices" / "libraries").rglob("*")))
    known = set(COMMON_ARABIC_WORDS) | set(DUPLICATE_LETTER_WHITELIST)
    for path in paths:
        if not path.is_file():
            continue
        try:
            raw = path.read_text(encoding="utf-8")
        except UnicodeError:
            continue
        if path.suffix.lower() == ".json":
            try:

            except json.JSONDecodeError:
                pass
        for word in ARABIC_WORD.findall(raw):
            normalized = normalize_word(word)
            if len(normalized) >= 2:

    return known


def lexicon_duplicate_whitelist() -> set[str]:
    known = set(DUPLICATE_LETTER_WHITELIST)
    for word in load_project_dictionary():
        base = normalize_word(word)
        if any(base[i] == base[i + 1] for i in range(len(base) - 1)):
            known.add(base)
    return known


def line_excerpt(line: str, start: int, end: int) -> str:
    left = max(0, start - 35)
    right = min(len(line), end + 35)
    return line[left:right].strip()


def find_punctuation_defects(text: str) -> dict[str, list[dict]]:
    result = defaultdict(list)
    for line_no, line in enumerate(text.splitlines(), 1):
        for token, label in (("، .", "فاصلة ثم نقطة"), (".،", "نقطة ثم فاصلة"),
                             ("،،", "فاصلتان"), ("..", "نقطتان")):
            for match in re.finditer(re.escape(token), line):
                result["علامات ترقيم متجاورة"].append({
                    "line": line_no, "sample": line_excerpt(line, match.start(), match.end()),
                    "reason": label,
                })
        for match in SPACE_BEFORE_PUNCT.finditer(line):
            result["مسافة قبل علامة ترقيم"].append({
                "line": line_no, "sample": line_excerpt(line, match.start(), match.end()),
                "reason": f"مسافة قبل «{match.group(1)}»",
            })
    return result


def find_duplicate_letters(text: str) -> list[dict]:
    whitelist = lexicon_duplicate_whitelist()
    found = []
    for line_no, line in enumerate(text.splitlines(), 1):
        for match in ARABIC_WORD.finditer(line):
            raw = match.group()
            word = normalize_word(raw)
            if len(word) < 2:
                continue
            positions = [i for i in range(len(word) - 1) if word[i] == word[i + 1]]
            # «يةان» تركيب صرفي غير سليم: يسبق اللاحقة الدالة على المثنى حرف التاء.
            if word.endswith("يةان"):
                found.append({"line": line_no, "word": raw, "reason": "لاحقة «يةان» غير سليمة صرفيًا"})
                continue
            if not positions or word in whitelist:
                continue
            for pos in positions:
                repeated = word[pos : pos + 2]
                # البادئات والتضعيفات الموثقة في القائمة البيضاء لا تُعد عيبًا.
                if pos == 0 and repeated in {"لل", "مم", "تت", "وو", "بب"}:
                    continue
                # المانع: تسلسلات لا تصحّ عربيًا («ةة» · «اا» · «وو» · «ىى» · «ئئ» · «ؤؤ»)
                # أو حرف تكرّر ثلاث مرات. أما تضاعف حرفين في كلمة عربية سليمة
                # («عدده» · «مخطط» · «وتحقق» · «وتغيير») فتنبيهي لا مانع — تصحيح ك-134.
                blocking = (repeated in {"ةة", "اا", "وو", "ىى", "ئئ", "ؤؤ"}
                            or word.count(word[pos]) >= 3)
                found.append({
                    "line": line_no, "word": raw, "blocking": blocking,
                    "reason": f"تكرار الحرف «{word[pos]}» داخل الكلمة"
                              + ("" if blocking else " — مشروع عربيًا، تنبيهي"),
                })
                break
    return found

def normalize_sentence(sentence: str) -> str:
    sentence = strip_markup(sentence).lower()
    sentence = re.sub(r"[0-90-9]+", " ", sentence)
    sentence = re.sub(r"https?://\S+|[A-Za-z][A-Za-z0-9_./-]*", " ", sentence)
    # الأسماء المقتبسة وأسماء السفن تُحذف كي يبقى قالب الجملة.
    sentence = re.sub(r"[«“\"].*?[»”\"]", " ", sentence)
    sentence = re.sub(r"#[\w-]+", " ", sentence)
    sentence = re.sub(r"\b(?:الأول|الثاني|الثالث|الرابع|الخامس|السادس|السابع|الثامن|التاسع|العاشر)\b", " ", sentence)
    # ما بعد الشرطة في قالب التوجيه يختلف باختلاف الوجهة، لا باختلاف القالب.
    sentence = re.sub(r"\bنفّذوا\s*[—–-].*", "نفّذوا —", sentence)
    sentence = re.sub(r"\s+", " ", sentence).strip(" ،؛:")
    return sentence


def find_repeated_templates(text: str) -> list[dict]:
    candidates = []
    for line_no, line in enumerate(text.splitlines(), 1):
        if "|" in line:
            continue  # أسطر الجداول: تكرار الخلية بنية جدول لا قالب جملة (معايرة 23 أكتوبر)
        clean_line = strip_markup(line).strip()
        if not clean_line or clean_line.startswith("#"):
            continue
        # bullets تمثل جملًا مستقلة حتى إن لم تنته بعلامة ترقيم.
        for sentence in re.split(r"(?<=[.!؟؛])\s+|\s*\n\s*", clean_line):
            normalized = normalize_sentence(sentence)
            has_marker = any(marker in normalized for marker in ("نفّذوا", "بلاغ من", "لا بلاغ جديد"))
            has_sentence_end = sentence.rstrip().endswith((".", "؟", "!", "؛"))
            if len(normalized) >= 4 and (has_marker or has_sentence_end):
                candidates.append((normalized, line_no, sentence.strip()))
    groups: dict[str, list[dict]] = defaultdict(list)
    for normalized, line_no, sample in candidates:
        groups[normalized].append({"line": line_no, "sample": sample[:120]})
    return [
        {"template": template, "count": len(items), "occurrences": items}
        for template, items in sorted(groups.items())
        if len(items) > 2
    ]


def find_unknown_words(text: str) -> list[str]:
    known = load_project_dictionary()
    words = Counter()
    for raw in ARABIC_WORD.findall(strip_markup(text)):
        word = normalize_word(raw)
        if len(word) < 3 or word in known:
            continue
        # الكلمات ذات الأعلام والأرقام والمصطلحات الأجنبية لا تدخل هذا التحذير.
        if any(ch.isdigit() for ch in raw):
            continue
        words[word] += 1
    return [word for word, _ in words.most_common()]

"""بوابة مانعة لتسرب عمل التحرير الداخلي إلى التقرير أو واجهة الموقع."""

    (re.compile(r"الخط\s+اليومي"), "عبارة عمل داخلي: الخط اليومي"),

    (re.compile(r"سجل\s+الدروس"), "عبارة عمل داخلي: سجل الدروس"),
    (re.compile(r"(?:^|[\s`(])tools/[^\s`),؛]+"), "اسم أداة داخلي: tools/…"),
    (re.compile(r"(?:^|[\s`(])validators/[^\s`),؛]+"), "اسم مدقق داخلي: validators/…"),
    (re.compile(r"(?<![\w])(?:[A-Za-z0-9_-]+/)?[^\s`),؛]+\.py\b"), "امتداد بايثون داخلي: .py"),
    (re.compile(r"(?<![\w])(?:[A-Za-z0-9_-]+/)?[^\s`),؛]+\.json\b"), "امتداد بيانات داخلي: .json"),
]


def body_lines(text: str):
    """يعيد أسطر المتن، مع إسقاط front matter والكتل البرمجية."""
    lines = text.splitlines()
    in_front = bool(lines and lines[0].strip() == "---")
    in_fence = False
    for number, line in enumerate(lines, 1):
        stripped = line.strip()
        if in_front:
            if number > 1 and stripped == "---":
                in_front = False
            continue
        if stripped.startswith("#"):
            continue  # العناوين ليست متنًا
        if stripped.startswith("- المصادر:") or stripped.startswith("- تشغيل منقوص مُفصح:"):
            continue  # أسطر مُولَّدة آليًا ومعتمدة في المتن
        if stripped.startswith("```") or stripped.startswith("~~~"):
            in_fence = not in_fence
            continue
        if not in_fence and stripped:
            yield number, line


def scan_text(text: str, source: str):
    findings = []
    for line_no, line in body_lines(text):
        for pattern, reason in INTERNAL_PATTERNS:
            for match in pattern.finditer(line):
                sample = match.group(0).strip(" `()؛،")
                findings.append({
                    "source": source,
                    "line": line_no,
                    "reason": reason,
                    "match": sample,
                    "text": line.strip(),
                })
    return findings


def scan_json(path: Path):
    # التحقق من JSON أولًا، ثم فحص تمثيله النصي كي يظهر موضع السطر للمراجع.
    raw = path.read_text(encoding="utf-8")
    json.loads(raw)
    return scan_text(raw, str(path))


def main() -> int:
    parser = argparse.ArgumentParser(description="بوابة التسرب الداخلي من التقرير")
    parser.add_argument("path", nargs="?", help="مسار التقرير")
    parser.add_argument("--md", dest="md", help="مسار تقرير Markdown")
    parser.add_argument("--meta", help="ملف واجهة JSON اختياري للفحص")
    args = parser.parse_args()
    chosen = args.md or args.path
    if not chosen and not args.meta:
        print("الاستخدام: check_internal_leak.py --md <التقرير> [--meta <الواجهة>]")
        return 2
    findings = []
    if chosen:
        path = Path(chosen)
        if not path.exists():
            print(f"الملف غير موجود: {path}")
            return 2
        findings.extend(scan_text(path.read_text(encoding="utf-8"), str(path)))
    if args.meta:
        meta = Path(args.meta)
        if not meta.exists():
            print(f"ملف الواجهة غير موجود: {meta}")
            return 2
        try:
            findings.extend(scan_json(meta))
        except json.JSONDecodeError as exc:
            print(f"ملف الواجهة JSON غير صالح: {exc}")
            return 2
    print("=== بوابة منع تسرب المحتوى الداخلي ===")
    for item in findings:
        print(f"✘ {item['source']} — السطر {item['line']} — {item['reason']} — «{item['match']}»")
        print(f"  النص: {item['text']}")
    if findings:
        print(f"النتيجة: مانع — رُصد {len(findings)} تسرب داخلي.")
        return 2
    print("✔ النتيجة: سليم — لم يُرصد تسرب داخلي في الملفات المفحوصة.")
    return 0


if __name__ == "__main__":
    sys.path.insert(0, str(Path(__file__).resolve().parent))
    from _guard import run
    run(main)

"""أرقام المصادر الثلاثة — مصدر واحد للحقيقة (ح-3 · قرار صاحب القرار 2026-10-23).

def inject(date: str, c: dict) -> tuple[bool, str]:
    """يكتب سطر الأرقام وسطر الإفصاح داخل قسم «ما لم يُحسم» (ظاهر في PDF) لا في الملحق وحده.

    الكتابة متكررة بلا تكرار: أي سطر مُولَّد سابقًا يبدأ بـ«- المصادر:» أو
    «- تشغيل منقوص مُفصح:» يُستبدل. لا وسوم تعليق في المتن — وسم التعليق ظهر نصًّا
    للقارئ في بناء الموقع (ك-135)، فاستُبدل ببادئة نصية ثابتة.
    """
    p = report_path(date)
    if not p.exists():
        return False, f"لا نص تقرير: {p}"
    t = p.read_text(encoding="utf-8")
    fl = faults_line(date)
    gen = [f"- {line_ar(c)}"] + ([f"- {fl}"] if fl else [])
    # إزالة النسخ المُولَّدة سابقًا في كل الملف
    t = re.sub(r"^-\s*(?:المصادر:|تشغيل منقوص مُفصح:).*\n?", "", t, flags=re.M)
    t = re.sub(r"^.*<!--\s*/?مصادر-تلقائي\s*-->.*\n?", "", t, flags=re.M)
    m = re.search(r"^##\s+ما لم يُحسم\s*$", t, re.M)
    if not m:
        return False, "قسم «ما لم يُحسم» غير موجود — لا يُكتب الرقم في مكان غير معتمد"
    nxt = re.search(r"^##\s+", t[m.end():], re.M)
    end = m.end() + (nxt.start() if nxt else len(t[m.end():]))
    body = t[m.end():end].rstrip("\n")
    t = t[:m.end()] + "\n" + body + "\n\n" + "\n".join(gen) + "\n" + t[end:]
    p.write_text(t, encoding="utf-8")
    return True, str(p)

"""تصنيف الأعطال المعروفة — قاعدة آلية لا حكم بشري (قرار صاحب القرار 2026-10-23 · ح-4).

كل عطل في التشغيل يُصنَّف: إمّا معروف مسموح بإفصاحه (وفق appendices/known-faults.json)،

from __future__ import annotations

import json

def expired(item: dict, today: str | None = None) -> bool:
    """البند المنتهي صلاحيته لا يُفصح عنه: يصير عطلًا غير مسموح فيمنع التسليم."""
    exp = (item or {}).get("expires")
    if not exp:
        return False
    import datetime as _dt
    d = today or _dt.datetime.now(_dt.timezone.utc).strftime("%Y-%m-%d")
    return exp < d


def load() -> dict:
    return json.loads(CONFIG.read_text(encoding="utf-8"))


def _match(entry: dict, component: str, text: str) -> bool:
    if expired(entry):
        return False  # البند المنتهي صلاحيته لا يُفصح عنه ⇒ يبقى العطل غير مسموح فيمنع
    if component not in (entry.get("applies_to") or []):
        return False
    pats = entry.get("reason_patterns") or []
    return any(p.lower() in text.lower() for p in pats)


def classify_component(component: str, status: str, text: str, items: list[str] | None = None) -> dict:
    """يصنّف عطل مكوّن واحد.

    items: قائمة نصوص الأعطال المفردة (مثل أسباب تعذّر التغذيات) — تُصنَّف فردًا فردًا.
    يعيد: {allowed, entry_id, reason_ar, uncovered: [نصوص لم تُغطَّ], covered: int}
    """
    cfg = load()
    items = items or []
    if status == "ok":
        return {"allowed": True, "entry_id": None, "reason_ar": "", "reason_ar_full": "",
                "uncovered": [], "covered": 0}

    if items:
        covered = 0
        uncovered: list[str] = []
        entry_id = None
        reason_ar = ""
        per_entry: dict[str, int] = {}
        for it in items:
            hit = next((e for e in cfg["allowed"] if _match(e, component, it)), None)
            if not hit:
                uncovered.append(it)
                continue
            per_entry[hit["id"]] = per_entry.get(hit["id"], 0) + 1
            cap = hit.get("max_items")
            if cap is not None and per_entry[hit["id"]] > int(cap):
                uncovered.append(f"{it} (تجاوز حدّ «{hit['id']}» = {cap})")
                continue
            covered += 1
            entry_id = entry_id or hit["id"]
            reason_ar = reason_ar or hit["reason_ar"]
        hit0 = next((e for e in cfg["allowed"] if e["id"] == entry_id), None) if entry_id else None
        return {"allowed": not uncovered, "entry_id": entry_id, "reason_ar": reason_ar,
                "reason_ar_full": (hit0.get("reason_ar_full") if hit0 else reason_ar) or reason_ar,
                "uncovered": uncovered, "covered": covered}

    hit = next((e for e in cfg["allowed"] if _match(e, component, f"{status} {text}")), None)
    if hit:
        return {"allowed": True, "entry_id": hit["id"], "reason_ar": hit["reason_ar"],
                "reason_ar_full": hit.get("reason_ar_full") or hit["reason_ar"],
                "uncovered": [], "covered": 1}
    return {"allowed": False, "entry_id": None, "reason_ar": "", "reason_ar_full": "",
            "uncovered": [f"{component}: {status} — {text[:120]}"], "covered": 0}


def degraded_components(mark: dict) -> list[tuple[str, str, str, list[str]]]:
    """يستخرج كل مكوّن حالته ليست ok مع نصّه وأعطاله المفردة."""

    SKIP = {"final_gate", "sources", "written_stage", "ledger", "pipeline"}
    for k, v in (mark or {}).items():
        if k in SKIP or not isinstance(v, dict) or "status" not in v:
            continue
        st = str(v.get("status"))
        if st == "ok":
            continue
        items: list[str] = []
        if k == "feeds":
            for f in (v.get("failed_feeds") or []):
                if isinstance(f, dict):
                    items.append(str(f.get("block_reason") or f.get("reason") or f.get("name") or f))
                else:
                    items.append(str(f))
        out.append((k, st, str(v.get("note") or v.get("lasterror") or ""), items))
    return out

#!/usr/bin/env python3
"""خطوة الإصلاح الآلي الواحدة للصياغة — قبل إعادة البوابة (طلب صاحب القرار، البند 8).

تعمل لغةً وترقيمًا فقط: لا تمسّ الأرقام ولا الرموز ولا الوقائع.
تصلح: الألفاظ المصطنعة المعتمدة، والقالب المتكرر أكثر من الحد، والترقيم المتجاور والمسافة قبله.
تخرج: 0 = لا مخالفة بعد الإصلاح · 2 = بقيت مخالفات ⇒ التسليم يوقف.
"""
from __future__ import annotations

import argparse
import json
import re
import subprocess
import sys
from pathlib import Path

B = Path(__file__).resolve().parent.parent
RULES = B / "appendices" / "writing-rules.json"


def load_rules() -> dict:
    return json.loads(RULES.read_text(encoding="utf-8"))


def fix_punctuation(text: str) -> tuple[str, int]:
    n = 0
    pats = [(r"\s+([،؛:.!؟])", r"\1"), (r"([،؛])\1+", r"\1"), (r"([،؛])\s*([.،؛])", r"\1")]
    for pat, rep in pats:
        text, k = re.subn(pat, rep, text)
        n += k
    return text, n


def fix_terms(text: str, rules: dict) -> tuple[str, int]:
    n = 0
    for item in rules.get("autoreplace", []):
        term, rep = item["term"], item["replacement"]
        text, k = re.subn(rf"(?<![\w\u0600-\u06FF]){re.escape(term)}(?![\w\u0600-\u06FF])", rep, text)
        n += k
    return text, n


def fix_templates(text: str, rules: dict) -> tuple[str, int]:
    cap = int(rules.get("template_cap", 2))
    alts = rules.get("template_alternatives") or []
    if not alts:
        return text, 0
    lines = text.splitlines()
    # القالب = بداية سطر نقطي ثابتة («- **نفّذوا** — ») تتكرر أكثر من الحد
    heads: dict[str, list[int]] = {}
    for i, line in enumerate(lines):
        m = re.match(r"^(\s*[-*]\s*\*\*[^*]+\*\*\s*[—–-])\s", line)
        if m:
            heads.setdefault(m.group(1).strip(), []).append(i)
    n = 0
    for head, idxs in heads.items():
        if len(idxs) <= cap:
            continue
        for j, i in enumerate(idxs[cap:]):
            alt = alts[j % len(alts)]
            rest = lines[i].split(head, 1)[1].lstrip()
            lines[i] = alt + rest
            n += 1
    return "\n".join(lines) + ("\n" if text.endswith("\n") else ""), n


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--date", required=True)
    ap.add_argument("--md", default=None)
    ap.add_argument("--dry-run", action="store_true")
    a = ap.parse_args()
    md = Path(a.md) if a.md else B / "reports" / f"{a.date}-sectioned.md"
    if not md.exists():
        print(f"  ✘ لا نص تقرير: {md}")
        return 2
    rules = load_rules()
    text = md.read_text(encoding="utf-8")
    before = text
    text, p1 = fix_punctuation(text)
    text, p2 = fix_terms(text, rules)
    text, p3 = fix_templates(text, rules)
    changed = text != before
    if changed and not a.dry_run:
        md.write_text(text, encoding="utf-8")
    print(f"  · الترقيم: {p1} · الألفاظ: {p2} · القوالب المُنوّعة: {p3}"
          + ("  (تجربة بلا كتابة)" if a.dry_run else ""))
    chk = subprocess.run([sys.executable, str(B / "validators" / "check_prose_defects.py"), "--md", str(md)],
                         capture_output=True, text=True)
    rc = chk.returncode
    tail = [l for l in chk.stdout.splitlines() if "النتيجة" in l]
    print("  " + (tail[-1] if tail else f"رمز المدقق: {rc}"))
    if rc >= 2:   # 1 = تنبيه (كلمات مشتبهة) لا يمنع · 2 = مخالفة مانعة

        return 2
    print("  ✔ لا مخالفة صياغة بعد الإصلاح.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

IMO_RE = re.compile(r'\bIMO\s*(?:NO\.?|NUMBER|#)?\s*([0-9]{7})\b', re.IGNORECASE)

# قائمة معلنة لكلمات وظيفية شائعة في أسماء الشركات؛ لا تُحذف أي كلمة أخرى بسبب طولها.
NAME_STOP_WORDS = frozenset({
    'AND', 'CO', 'COMPANY', 'CORP', 'GROUP', 'INC', 'INTERNATIONAL', 'LTD', 'LLC',
    'LINE', 'LINES', 'MARINE', 'MARITIME', 'MV', 'MT', 'OF', 'SA', 'SHIP', 'SHIPPING',
    'THE', 'TRADING',
})
NAME_PARTIAL_MIN_SCORE = 75


def extract_imos(value: str) -> set[str]:
    """يستخرج أرقام IMO ذات السبعة أرقام من أي حقل نصي."""
    return {m.group(1) for m in IMO_RE.finditer(str(value or ''))}


def _name_tokens(value: str) -> set[str]:
    """يعيد كلمات الاسم بعد حذف كلمات الوظيفة المعلنة فقط."""
    without_imo = IMO_RE.sub(' ', str(value or '').upper())
    tokens = re.findall(r'[A-Z0-9]+', without_imo)
    return {token for token in tokens if token not in NAME_STOP_WORDS and token != 'IMO'}


def _normalise_entry(entry: Any) -> dict[str, Any]:
    """يوحّد إدخالات القائمة، مع قبول اسم نصي للتوافق مع المستعملين السابقين."""
    if isinstance(entry, str):
        return {'name': entry, 'imos': extract_imos(entry)}
    if not isinstance(entry, dict):
        return {'name': str(entry), 'imos': set()}
    name = str(entry.get('name') or entry.get('sdn_name') or '').strip()
    imos: set[str] = set()
    for key in ('imos', 'imo', 'imo_number'):
        value = entry.get(key)
        if isinstance(value, (list, tuple, set)):
            for item in value:
                text = str(item)
                imos |= extract_imos(text)
                if re.fullmatch(r'\d{7}', text.strip()):
                    imos.add(text.strip())
        elif value:
            text = str(value)
            imos |= extract_imos(text)
            if re.fullmatch(r'\d{7}', text.strip()):
                imos.add(text.strip())

def _name_score(target: str, candidate: str) -> tuple[str | None, int]:
    target_tokens = _name_tokens(target)
    candidate_tokens = _name_tokens(candidate)
    if not target_tokens or not candidate_tokens:
        return None, 0
    shared = target_tokens & candidate_tokens
    if target_tokens == candidate_tokens:
        return 'name_full', 100
    # مقياس F1 يمنع اسمًا من كلمة واحدة من مطابقة اسم من كلمتين بكلمة مشتركة.
    if len(shared) < 2:
        return None, round(100 * (2 * len(shared) / (len(target_tokens) + len(candidate_tokens))))
    score = round(100 * (2 * len(shared) / (len(target_tokens) + len(candidate_tokens))))
    return ('name_partial', score) if score >= NAME_PARTIAL_MIN_SCORE else (None, score)


def _detail(entry: dict[str, Any], match_type: str, score: int) -> dict[str, Any]:
    return {
        'name': entry['name'],
        'type': match_type,
        'score': score,
        'percentage': f'{score}%',
    }


def _result(target: str, result: str, details: list[dict[str, Any]]) -> dict[str, Any]:
    first = details[0] if details else None
    return {
        'target': target,
        'match': [d['name'] for d in details[:5]],
        'match_details': details[:5],
        'match_type': first['type'] if first else None,
        'match_score': first['score'] if first else None,
        'match_percentage': first['percentage'] if first else None,
        'result': result,
    }


def match_target(target: str, entries: list[Any]) -> dict[str, Any]:
    """يفحص مستهدفًا: IMO أولًا، ثم الاسم الكامل، ثم الاسم الجزئي عالي النسبة."""
    normalised = [_normalise_entry(entry) for entry in entries]
    target_imos = extract_imos(target)
    imo_hits = [entry for entry in normalised if target_imos & entry['imos']]
    if imo_hits:
        return _result(target, 'match', [_detail(entry, 'imo', 100) for entry in imo_hits])

    full_hits, partial_hits = [], []
    for entry in normalised:
        match_type, score = _name_score(target, entry['name'])
        if match_type == 'name_full':
            full_hits.append(_detail(entry, match_type, score))
        elif match_type == 'name_partial':
            partial_hits.append(_detail(entry, match_type, score))
    if full_hits:
        return _result(target, 'match', full_hits)
    if partial_hits:
        partial_hits.sort(key=lambda detail: (-detail['score'], detail['name']))
        return _result(target, 'review', partial_hits)
    return _result(target, 'no_match', [])


def load() -> tuple[str | None, list[dict[str, Any]]]:
    """يحمّل أسماء القائمة وأرقام IMO الموجودة في صفوفها."""
    for url in URLS:
        try:
            raw = urllib.request.urlopen(urllib.request.Request(url, headers=UA), timeout=60).read()
            text = raw.decode('utf-8', 'ignore')
            entries = []
            for row in csv.reader(io.StringIO(text)):
                fields = [field.strip() for field in row if field.strip()]
                if not fields:
                    continue
                name = next((field for field in fields[1:] if not field.isdigit()), fields[0])
                entries.append({'name': name.upper(), 'imos': sorted(extract_imos(' '.join(fields)))})
            if entries:
                return url, entries
        except Exception as e:
            print(f"  تعذر {url[:60]} → {str(e)[:60]}")
    return None, []


def partial_strength(vessel: str, names: list[str]) -> tuple[str, str]:
    """يحافظ على واجهة التصنيف القديمة مع تطبيق قاعدة الاسم الموزون."""
    details = []
    for name in names:
        match_type, score = _name_score(vessel, name)
        if match_type == 'name_partial':
            details.append(f'{name} ({score}%)')
    return ('warning', ' · '.join(details)) if details else ('info', ' · '.join(names))
